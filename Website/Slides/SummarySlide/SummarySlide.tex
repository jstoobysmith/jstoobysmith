        \vspace{3pt}
\begin{lstlisting}
def oddSum (n : Nat) : Nat :=
  (Finset.range n).sum
    (fun i => 2 * i + 1)

theorem oddSum_eq (n : Nat) :
    oddSum n = n ^ 2 := by
  induction n with
  | zero => simp [oddSum]
  | succ n ih =>
    unfold oddSum at *
    rw [Finset.sum_range_succ, ih]
    ring
\end{lstlisting}
      \end{plplain}
    \end{column}
