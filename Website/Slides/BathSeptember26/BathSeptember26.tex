-- "the integers"
def MyInt : Type := Empty

def add (a b : MyInt) : MyInt :=
  Empty.elim a
def pow (a : MyInt) (n : Nat) : MyInt :=
  Empty.elim a

theorem fermat_last_theorem (a b c : MyInt)
    (n : Nat) (hn : 3 <= n) :
    Ne (add (pow a n) (pow b n)) (pow c n) :=
  Empty.elim a
\end{lstlisting}
